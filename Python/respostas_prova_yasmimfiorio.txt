#Comprando até 3 livros: desconto 8%

PRECO_LIVRO = 39.90
sem_desconto = 0
com_desconto = 0

if quantidade == 3:
    sem_desconto = PRECO_LIVRO + quantidade %8
    print("Desconto de 8%")

else:
